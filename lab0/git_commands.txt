git status - посмотреть состояние файлов
git add - добавить изменения в историю
git commit - сохранить изменения в историю
git cherry-pick - переносит конкретный коммит из одной ветки в другую
git checkout - переключиться на другую ветку
git clone - скачать репозиторий с гитхаб
git merge - объединяет две ветки
git reset - отменяет изменения
git branch - посмотреть или создать ветку
git push - отправить сохранение в гитхаб
